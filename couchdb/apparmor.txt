#include <tunables/global>

profile couchdb flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/bash>
  #include <abstractions/nameservice>
  #include <abstractions/openssl>

  # Baseline profile for CouchDB. CouchDB runs on the Erlang VM, which needs
  # broad file and network access plus the ability to drop privileges to the
  # couchdb user.
  file,
  network,

  capability chown,
  capability dac_override,
  capability fowner,
  capability setgid,
  capability setuid,

  # runc/docker stops the container by signalling PID 1. Without receive
  # permission the signal is denied ("apparmor=DENIED operation=signal
  # signal=kill peer=runc"), docker stop escalates straight to SIGKILL, and
  # CouchDB never shuts down cleanly.
  signal (receive) peer=unconfined,
  signal (receive) peer=runc,

  deny /proc/kcore rwklx,
  deny /proc/sysrq-trigger rwklx,
  deny /sys/firmware/** rwklx,
}
